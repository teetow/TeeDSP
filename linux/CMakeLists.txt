cmake_minimum_required(VERSION 3.21)
project(TeeDSPHeadless LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
include(FetchContent)
FetchContent_Declare(clap GIT_REPOSITORY https://github.com/free-audio/clap.git GIT_TAG 1.2.2 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(clap)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3)
pkg_check_modules(HTTP REQUIRED IMPORTED_TARGET cpp-httplib)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)
add_library(teedsp_core STATIC
  ${ROOT}/src/dsp/Compressor.cpp ${ROOT}/src/dsp/Exciter.cpp
  ${ROOT}/src/dsp/Leveler.cpp ${ROOT}/src/dsp/SpectralLeveler.cpp
  ${ROOT}/src/dsp/ParametricEQ.cpp ${ROOT}/src/dsp/ProcessorChain.cpp
  ${ROOT}/plugin/TeeDspPlugin.cpp ${ROOT}/src/host/Fft.cpp)
target_include_directories(teedsp_core PUBLIC ${ROOT}/src ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(teedsp_core PUBLIC clap nlohmann_json::nlohmann_json)
add_executable(teedsp main.cpp)
target_link_libraries(teedsp PRIVATE teedsp_core PkgConfig::PIPEWIRE PkgConfig::HTTP Threads::Threads)
add_executable(headless-tests tests/processor.cpp)
target_link_libraries(headless-tests PRIVATE teedsp_core)
add_test(NAME headless-tests COMMAND headless-tests)
add_executable(leveler-tests ${ROOT}/tests/LevelerTests.cpp)
target_link_libraries(leveler-tests PRIVATE teedsp_core)
add_test(NAME leveler-tests COMMAND leveler-tests)
